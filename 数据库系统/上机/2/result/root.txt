mysql> # 创建视图 branch_detail，用于汇总支行的存款/贷款数据
mysql> CREATE VIEW branch_detail AS
    -> SELECT 
    ->     b.branch_name,  -- 从支行表获取支行名称
    ->     COALESCE(depositor_count, 0) AS depositor_count,  -- 存款客户数量（无数据时显示0）
    ->     COALESCE(total_balance, 0) AS total_balance,     -- 存款总额（无数据时显示0）
    ->     COALESCE(borrower_count, 0) AS borrower_count,   -- 贷款客户数量（无数据时显示0）
    ->     COALESCE(total_loan, 0) AS total_loan            -- 贷款总额（无数据时显示0）
    -> FROM branch b  -- 主表：所有支行
    -> -- 左连接存款数据（按支行分组聚合）
    -> LEFT JOIN (
    ->     SELECT 
    ->         branch_name,  -- 分组依据：支行名称
    ->         COUNT(DISTINCT customer_name) AS depositor_count,  -- 去重统计客户数
    ->         SUM(balance) AS total_balance  -- 汇总余额
    ->     FROM account a  -- 存款账户表
    ->     JOIN depositor d ON a.account_number = d.account_number  -- 关联存款人表
    ->     GROUP BY branch_name  -- 按支行分组
    -> ) depositor_data ON b.branch_name = depositor_data.branch_name  -- 连接条件
    -> -- 左连接贷款数据（按支行分组聚合）
    -> LEFT JOIN (
    ->     SELECT 
    ->         branch_name,  -- 分组依据：支行名称
    ->         COUNT(DISTINCT customer_name) AS borrower_count,  -- 去重统计客户数
    ->         SUM(amount) AS total_loan  -- 汇总贷款金额
    ->     FROM loan l  -- 贷款表
    ->     JOIN borrower br ON l.loan_number = br.loan_number  -- 关联借款人表
    ->     GROUP BY branch_name  -- 按支行分组
    -> ) borrower_data ON b.branch_name = borrower_data.branch_name;  -- 连接条件
Query OK, 0 rows affected (0.01 sec)
